// 1. إعدادات الحاوية السحابية (JSONBin) const BIN_ID = "6aa17373ffd5d16053f2232a"; const API_URL = `https://api.jsonbin.io/v3/b/${BIN_ID}`; // مفتاح الوصول API Key (تحتاج الحصول عليه مجاناً من حسابك في jsonbin.io) // يمكنك استبداله بمفتاحك الشخصي X-Master-Key أو Master Key const MASTER_KEY = "$2a$10$YOUR_MASTER_KEY_HERE"; // 2. دالة جلب البيانات التلقائية من السحابة عند فتح الصفحة من أي جهاز async function loadDataFromCloud() { const statusText = document.getElementById('sync-status-text'); const statusDot = document.getElementById('sync-status-dot'); if (statusText) statusText.innerText = "جاري جلب البيانات من السحابة..."; try { const response = await fetch(`${API_URL}/latest`, { method: 'GET', headers: { 'X-Master-Key': MASTER_KEY } }); if (!response.ok) throw new Error("فشل الاتصال بالسحابة"); const data = await response.json(); // تحديث مصفوفات البرنامج بالبيانات الجاهزة من السحابة if (data.record) { window.appData = data.record; renderAll(); // إعادة رسم كافة الجداول واللوحات } if (statusText && statusDot) { statusText.innerText = "السحابة متصلة ومحدثة ⚡"; statusDot.className = "inline-block w-2.5 h-2.5 bg-emerald-400 rounded-full animate-pulse"; } } catch (error) { console.error("خطأ في جلب البيانات:", error); if (statusText) statusText.innerText = "تعذر الاتصال بالسحابة (يعمل أوفلاين)"; } } // 3. دالة حفظ أي تعديل وإرساله فوراً للسحابة لتشاهده جميع الأجهزة async function saveToCloud() { const statusText = document.getElementById('sync-status-text'); if (statusText) statusText.innerText = "جاري حفظ التعديل على السحابة..."; try { const response = await fetch(API_URL, { method: 'PUT', headers: { 'Content-Type': 'application/json', 'X-Master-Key': MASTER_KEY }, body: JSON.stringify(window.appData) }); if (response.ok) { if (statusText) statusText.innerText = "تم الحفظ المباشر عبر كافة الأجهزة ⚡"; renderAll(); } else { alert("حدث خطأ أثناء الحفظ السحابي، يُرجى التأكد من الـ Master Key"); } } catch (error) { console.error("خطأ في الحفظ السحابي:", error); alert("تعذر الاتصال بالسيرفر لحفظ التعديلات."); } } // 4. تشغيل جلب البيانات تلقائياً بمجرد فتح الصفحة من الموبايل أو الكمبيوتر document.addEventListener('DOMContentLoaded', () => { loadDataFromCloud(); });
